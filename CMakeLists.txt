cmake_minimum_required(VERSION 3.16)
project(ksl VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(ksl src/main.cpp)
if(MSVC)
  target_compile_options(ksl PRIVATE /W4)
else()
  target_compile_options(ksl PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME language COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/run.py $<TARGET_FILE:ksl>)
endif()
